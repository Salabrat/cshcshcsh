Manual Slider Fix for Desktop

If the trending products slider is showing white space after the last product on desktop, use the buttons below to fix it:

Or copy this code to browser console:

// Fix slider position (function() { const grid = document.querySelector('.trending-grid'); const items = document.querySelectorAll('.trending-item'); const itemsPerView = window.innerWidth > 768 ? 4 : 2; const maxIndex = Math.max(0, items.length - itemsPerView); let currentIndex = parseInt(grid.dataset.currentIndex) || 0; if (currentIndex > maxIndex) { currentIndex = maxIndex; const translateX = -(currentIndex * 25); // 25% per item for 4 items view grid.style.transform = `translateX(${translateX}%)`; grid.dataset.currentIndex = currentIndex; console.log('Fixed slider to index:', currentIndex); } })();
